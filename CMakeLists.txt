cmake_minimum_required(VERSION 3.24)
project(KernelLab VERSION 0.1.0 LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "KernelLab targets Linux. Use native Linux or WSL2, not a Windows toolchain.")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build: cmake --preset cpu-debug")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")

option(KERNELLAB_ENABLE_CUDA "Build the CUDA backend (requires a CUDA toolkit)" ON)
option(KERNELLAB_BUILD_TOOLS "Build device_inspect" ON)
option(KERNELLAB_BUILD_BENCHMARKS "Build the benchmark entry point" ON)
option(KERNELLAB_BUILD_TRANSFORMER "Build the Tiny Transformer configuration skeleton" ON)
option(KERNELLAB_ENABLE_GPU_TESTS "Register GPU smoke tests; unavailable GPUs skip with code 77" OFF)
include(CTest)
include(cmake/ProjectOptions.cmake)

if(KERNELLAB_ENABLE_GPU_TESTS AND NOT KERNELLAB_ENABLE_CUDA)
  message(FATAL_ERROR "GPU tests require KERNELLAB_ENABLE_CUDA=ON.")
endif()

if(KERNELLAB_ENABLE_CUDA)
  include(cmake/EnableCUDA.cmake)
endif()

configure_file(cmake/build_config.h.in generated/gpulab/build_config.h @ONLY)
add_subdirectory(src)
if(KERNELLAB_BUILD_TOOLS)
  add_subdirectory(tools)
endif()
if(KERNELLAB_BUILD_BENCHMARKS)
  add_subdirectory(benchmarks)
endif()
if(KERNELLAB_BUILD_TRANSFORMER)
  add_subdirectory(transformer)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()

message(STATUS "KernelLab ${PROJECT_VERSION}: ${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}")
message(STATUS "CUDA=${KERNELLAB_ENABLE_CUDA}; tests=${BUILD_TESTING}; GPU tests=${KERNELLAB_ENABLE_GPU_TESTS}")
